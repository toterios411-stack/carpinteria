270    def guardar_usuario(self):
271        user = self.txt_nuevo_user.get().strip()
272        clave = self.txt_nueva_clave.get().strip()
273        rol_cargo = self.txt_rol.get().strip()
274        permisos = self.combo_permisos.get().strip()
275
276        if not user or not clave:
277            messagebox.showwarning("Atención", "El usuario y la contraseña son obligatorios.")
278            return
279
280        if not rol_cargo:
281            messagebox.showwarning("Atención", "El campo de Rol (Cargo) no puede estar vacío.")
282            return
283
284        if len(rol_cargo.split()) > 1 or " " in rol_cargo:
285            messagebox.showwarning("Atención", "El Rol (Cargo) debe ser estrictamente UNA SOLA PALABRA (sin espacios).")
286            return
287
288        rol_cargo_formateado = rol_cargo.upper()
289        rol_final = f"{rol_cargo_formateado} ({permisos})"
290
291        exito, msg = database.registrar_usuario(user, clave, rol_final)
292        
293        if exito:
294            messagebox.showinfo("Éxito", f"Usuario '{user}' registrado como '{rol_cargo_formateado}' con permisos de {permisos}.")
295            self.destroy()
296            if hasattr(self.master, "cargar_usuarios"):
297                self.master.cargar_usuarios()
298        else:
299            messagebox.showerror("Error", msg)
